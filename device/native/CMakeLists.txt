cmake_minimum_required(VERSION 3.21)
project(GoodBotBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick)
add_library(goodbotbridgeplugin MODULE BridgePlugin.cpp InkPad.cpp InkPad.h ControlEndpoint.cpp ControlEndpoint.h)
target_link_libraries(goodbotbridgeplugin PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
target_compile_options(goodbotbridgeplugin PRIVATE -Wall -Wextra -Werror)
set_target_properties(goodbotbridgeplugin PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/GoodBot/Bridge")
configure_file(qmldir "${CMAKE_BINARY_DIR}/GoodBot/Bridge/qmldir" COPYONLY)
install(TARGETS goodbotbridgeplugin LIBRARY DESTINATION GoodBot/Bridge)
install(FILES qmldir DESTINATION GoodBot/Bridge)
# Session.qml/Panel.qml and the application RCC are assembled by the device packager.

option(GOODBOT_BUILD_PROBE "Build the image/QML marshaling regression probe" OFF)
if(GOODBOT_BUILD_PROBE)
    add_executable(goodbot-native-probe BridgePlugin.cpp InkPad.cpp InkPad.h ControlEndpoint.cpp ControlEndpoint.h)
    target_compile_definitions(goodbot-native-probe PRIVATE GOODBOT_BRIDGE_PROBE)
    target_compile_options(goodbot-native-probe PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(goodbot-native-probe PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
    add_executable(goodbot-control-probe ControlEndpoint.cpp ControlEndpoint.h)
    target_compile_definitions(goodbot-control-probe PRIVATE GOODBOT_CONTROL_PROBE)
    target_compile_options(goodbot-control-probe PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(goodbot-control-probe PRIVATE Qt6::Core Qt6::Gui)
    enable_testing()
    add_test(NAME control_transport_journal COMMAND goodbot-control-probe)
    add_test(NAME image_qml_marshaling COMMAND goodbot-native-probe)
endif()

option(GOODBOT_BUILD_CONTROL_RELAY "Build the separate HTTPS native-control worker" OFF)
if(GOODBOT_BUILD_CONTROL_RELAY)
    find_package(CURL REQUIRED)
    add_executable(goodbot-control-relay ../control_relay.cpp)
    target_link_libraries(goodbot-control-relay PRIVATE Qt6::Core CURL::libcurl)
    target_compile_options(goodbot-control-relay PRIVATE -Wall -Wextra -Werror)
    enable_testing()
    add_test(NAME control_relay_validation COMMAND goodbot-control-relay --self-test)
endif()
